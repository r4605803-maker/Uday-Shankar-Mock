let currentIndex = 0;
let userAnswers = new Array(questions.length).fill(null);
let visited = new Array(questions.length).fill(false);
let timeSpent = new Array(questions.length).fill(0); // প্রশ্নভিত্তিক সময় (সেকেন্ড)
let lastSwitchTimestamp = null;
let candidateName = "";
let timeLeft = 25 * 60;
let timerInterval = null;

function startMockTest() {
  const nameInput = document.getElementById("candidate-name").value.trim();
  if (!nameInput) {
    alert("অনুগ্রহ করে তোমার নাম লিখো!");
    return;
  }
  candidateName = nameInput;
  document.getElementById("disp-candidate-name").innerText = "পরীক্ষার্থী: " + candidateName;
  document.getElementById("login-screen").style.display = "none";
  document.getElementById("exam-screen").style.display = "flex";

  renderTopBoxes();
  lastSwitchTimestamp = Date.now();
  loadQuestion(0);
  startTimer();
}

function renderTopBoxes() {
  const container = document.getElementById("boxes-grid");
  container.innerHTML = "";
  for (let i = 0; i < questions.length; i++) {
    const btn = document.createElement("div");
    btn.className = "box-btn";
    btn.id = "top-box-" + i;
    btn.innerText = (i + 1);
    btn.onclick = () => jumpToQuestion(i);
    container.appendChild(btn);
  }
}

function trackTimeOnCurrentQuestion() {
  if (lastSwitchTimestamp !== null) {
    const now = Date.now();
    const elapsedSec = Math.round((now - lastSwitchTimestamp) / 1000);
    timeSpent[currentIndex] += elapsedSec;
    lastSwitchTimestamp = now;
  }
}

function loadQuestion(index) {
  trackTimeOnCurrentQuestion();
  currentIndex = index;
  visited[index] = true;

  document.getElementById("q-title").innerText = `প্রশ্ন ${index + 1} / ${questions.length}`;
  document.getElementById("q-exam").innerText = questions[index].exam;
  document.getElementById("q-text").innerText = questions[index].q;

  const box = document.getElementById("options-box");
  box.innerHTML = "";

  questions[index].options.forEach((text, optIdx) => {
    const label = document.createElement("label");
    const radio = document.createElement("input");
    radio.type = "radio";
    radio.name = "ans_opt";
    radio.value = optIdx;
    if (userAnswers[index] === optIdx) radio.checked = true;
    radio.onchange = () => {
      userAnswers[index] = optIdx;
      updateTopBoxes();
    };

    label.appendChild(radio);
    label.appendChild(document.createTextNode(text));
    box.appendChild(label);
  });

  updateTopBoxes();
}

function updateTopBoxes() {
  for (let i = 0; i < questions.length; i++) {
    const el = document.getElementById("top-box-" + i);
    el.className = "box-btn";
    if (i === currentIndex) el.classList.add("active");

    if (userAnswers[i] !== null) {
      el.classList.add("answered"); // সবুজ
    } else if (visited[i]) {
      el.classList.add("visited"); // লাল
    }
  }
}

function nextQuestion() {
  if (currentIndex < questions.length - 1) {
    loadQuestion(currentIndex + 1);
  }
}

function prevQuestion() {
  if (currentIndex > 0) {
    loadQuestion(currentIndex - 1);
  }
}

function jumpToQuestion(index) {
  loadQuestion(index);
}

function clearOption() {
  userAnswers[currentIndex] = null;
  const radios = document.getElementsByName("ans_opt");
  radios.forEach(r => r.checked = false);
  updateTopBoxes();
}

function startTimer() {
  timerInterval = setInterval(() => {
    timeLeft--;
    const mins = Math.floor(timeLeft / 60);
    const secs = timeLeft % 60;
    document.getElementById("timer-display").innerText = 
      `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;

    if (timeLeft <= 0) {
      clearInterval(timerInterval);
      alert("সময় শেষ! তোমার পরীক্ষাটি অটো-সাবমিট হয়ে গেল।");
      finishExam();
    }
  }, 1000);
}

function confirmSubmit() {
  const answered = userAnswers.filter(a => a !== null).length;
  if (confirm(`তুমি ৩০ টির মধ্যে ${answered} টি প্রশ্নের উত্তর দিয়েছো। সাবমিট করতে চাও?`)) {
    clearInterval(timerInterval);
    finishExam();
  }
}

function formatDuration(sec) {
  if (sec < 60) return `${sec} সেকেন্ড`;
  const m = Math.floor(sec / 60);
  const s = sec % 60;
  return `${m} মিনিট ${s} সেকেন্ড`;
}

function finishExam() {
  trackTimeOnCurrentQuestion();
  document.getElementById("exam-screen").style.display = "none";
  document.getElementById("result-screen").style.display = "block";

  let correct = 0;
  let wrong = 0;
  const reviewList = document.getElementById("review-list");
  reviewList.innerHTML = `<h3 style="margin-bottom: 15px; color: var(--primary);">উত্তরপত্র এবং প্রতিটি প্রশ্নে ব্যয়িত সময়:</h3>`;

  questions.forEach((q, idx) => {
    const userChoice = userAnswers[idx];
    const isCorrect = userChoice === q.ans;
    if (userChoice !== null) {
      if (isCorrect) correct++;
      else wrong++;
    }

    const card = document.createElement("div");
    card.className = "review-card";

    let tagColor = "#64748b";
    let tagText = "উত্তর দেওনি";
    if (userChoice !== null) {
      tagColor = isCorrect ? "var(--success)" : "var(--danger)";
      tagText = isCorrect ? "সঠিক" : "ভুল";
    }

    card.innerHTML = `
      <div style="display:flex; justify-content:space-between; align-items:center; margin-bottom: 8px;">
        <div>
          <strong>প্রশ্ন ${idx + 1}</strong>
          <span class="time-badge">⏱ সময় লেগেছে: ${formatDuration(timeSpent[idx])}</span>
        </div>
        <span style="color: ${tagColor}; font-weight: bold;">${tagText}</span>
      </div>
      <p style="font-size: 14px; margin-bottom: 8px;">${q.q}</p>
      <div style="color: var(--success); font-weight: 600; font-size: 13.5px;">সঠিক উত্তর: ${q.options[q.ans]}</div>
      ${userChoice !== null && !isCorrect ? `<div style="color: var(--danger); font-weight: 600; font-size: 13.5px;">তোমার উত্তর: ${q.options[userChoice]}</div>` : ""}
    `;
    reviewList.appendChild(card);
  });

  document.getElementById("res-name").innerText = `${candidateName}, তোমার ফলাফল`;
  document.getElementById("res-correct").innerText = correct;
  document.getElementById("res-wrong").innerText = wrong;
  document.getElementById("res-total").innerText = `${correct} / ${questions.length}`;
}
